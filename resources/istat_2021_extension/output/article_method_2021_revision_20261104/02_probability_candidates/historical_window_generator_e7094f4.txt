def fit_quadratic_model(df, response_column):
    # Prepare the quadratic terms for the predictors 'age' and 'area'
    df['age_sq'] = df['age'] ** 2
    df['area_sq'] = df['area'] ** 2
    df['age_area'] = df['age'] * df['area']
    
    # Define the design matrix (X) and response (y)
    X = df[['age', 'area', 'age_sq', 'area_sq', 'age_area']]
    X = sm.add_constant(X)  # Adds the intercept (a0)
    y = df[response_column]
    
    # Fit the model
    model = sm.OLS(y, X).fit()
    
    # Extract the coefficients
    return model.params
def fit_linear_model(df, response_column):
    # Define the design matrix (X) and response (y)
    X = df[['age', 'area']]
    X = sm.add_constant(X)  # Adds the intercept (a0)
    y = df[response_column]
    
    # Fit the model
    model = sm.OLS(y, X).fit()
    
    # Extract the coefficients
    return model.params

def random_age_from_range(age_range):
    if isinstance(age_range, list) and len(age_range) == 2:
        return random.randint(age_range[0], age_range[1])
    return None

def convert_and_max(array):
    # Convert each element in the array to a float
    array=ast.literal_eval(array)
    numeric_array = np.array(array)
    return np.max(numeric_array)
def convert_and_count(array):
    # Convert each element in the array to a float
    array=ast.literal_eval(array)
    numeric_array = np.array(array)
    return np.count(numeric_array)
def convert_and_min(array):
    # Convert each element in the array to a float
    array=ast.literal_eval(array)
    numeric_array = np.array(array)
    return np.min(numeric_array)
